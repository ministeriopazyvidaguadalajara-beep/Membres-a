const SHEET_NAME = 'Miembros'; const COLS = ['id','nombre','apellidos','dni','fnac','sexo','civil','tel','email','dir','fconv','fbaut_ok','fbaut','ministerio','cargo','posicion','proc','falta','notas','estado']; function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(SHEET_NAME); if (!sh) sh = ss.insertSheet(SHEET_NAME); if (sh.getLastRow() === 0) sh.appendRow(COLS); var action = e && e.parameter && e.parameter.action; if (action === 'save') { var m = JSON.parse(e.parameter.data); guardarMiembro(sh, m); return json({ok: true}); } if (action === 'delete') { var id = e.parameter.id; borrarMiembro(sh, id); return json({ok: true}); } // Cargar todos los miembros var rows = sh.getDataRange().getValues(); if (rows.length < 2) return json([]); var headers = rows[0]; var data = rows.slice(1).map(function(r) { var o = {}; headers.forEach(function(h, i) { o[h] = r[i] !== undefined ? String(r[i]) : ''; }); return o; }); return json(data); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(SHEET_NAME); if (!sh) sh = ss.insertSheet(SHEET_NAME); if (sh.getLastRow() === 0) sh.appendRow(COLS); var payload = JSON.parse(e.postData.contents); if (payload.action === 'save') { guardarMiembro(sh, payload.member); return json({ok: true}); } if (payload.action === 'delete') { borrarMiembro(sh, payload.id); return json({ok: true}); } return json({ok: false, error: 'unknown action'}); } function guardarMiembro(sh, m) { var ids = sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow()-1, 1).getValues().map(function(r){return String(r[0]);}) : []; var idx = ids.indexOf(String(m.id)); var row = COLS.map(function(c) { return m[c] !== undefined ? m[c] : ''; }); if (idx === -1) { sh.appendRow(row); } else { sh.getRange(idx + 2, 1, 1, row.length).setValues([row]); } } function borrarMiembro(sh, id) { var ids = sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow()-1, 1).getValues().map(function(r){return String(r[0]);}) : []; var idx = ids.indexOf(String(id)); if (idx !== -1) sh.deleteRow(idx + 2); } function json(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
